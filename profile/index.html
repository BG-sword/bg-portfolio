dataproc_pyspark_pipeline.py
from pyspark.sql import SparkSession
from pyspark.sql.functions import col, when

# Initialize Spark for Cloud Dataproc
spark = SparkSession.builder \
    .appName("GCP_Lakehouse_ETL") \
    .config("spark.sql.parquet.compression.codec", "snappy") \
    .getOrCreate()

# Process CDC & SCD Type 2 Dimension Data
df_raw = spark.read.format("bigquery") \
    .load("enterprise_lake.raw_events")

df_transformed = df_raw.filter(col("status") == "SUCCESS") \
    .withColumn("latency_sec", col("end_ts") - col("start_ts"))

# Write Partitioned Output to BigQuery
df_transformed.write.format("bigquery") \
    .option("partitionField", "event_date") \
    .mode("append") \
    .save("enterprise_lakehouse.fact_events")
